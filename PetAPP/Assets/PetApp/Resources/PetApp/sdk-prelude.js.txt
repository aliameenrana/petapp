(function () {
  'use strict';

  var commands = [];
  var nextId = 1;
  var now = 0;
  var timers = [];
  var nextTimerId = 1;
  var tapHandlers = [];
  var dragHandlers = [];
  var touchHandlers = {};
  var offscreenHandlers = {};
  var seed = 1;
  var ended = false;

  function emit(command) {
    if (!ended) commands.push(command);
  }

  function num(v) {
    v = +v;
    return isFinite(v) ? v : 0;
  }

  function makeObject(id, x, y) {
    var pos = { x: x, y: y };
    var obj = {
      get x() { return pos.x; },
      set x(v) { pos.x = num(v); emit({ op: 'pos', id: id, x: pos.x, y: pos.y }); },
      get y() { return pos.y; },
      set y(v) { pos.y = num(v); emit({ op: 'pos', id: id, x: pos.x, y: pos.y }); },
      set velocity(v) { emit({ op: 'vel', id: id, x: num(v && v.x), y: num(v && v.y) }); },
      play: function (clip) { emit({ op: 'anim', id: id, s: String(clip) }); },
      destroy: function () {
        delete touchHandlers[id];
        delete offscreenHandlers[id];
        emit({ op: 'destroy', id: id });
      },
      onTouch: function (other, fn) {
        (touchHandlers[id] = touchHandlers[id] || []).push({ other: other.id, fn: fn });
      },
      onOffscreen: function (fn) { offscreenHandlers[id] = fn; }
    };
    Object.defineProperty(obj, 'id', { value: id });
    return obj;
  }

  function addTimer(delay, fn, repeat) {
    var t = { id: nextTimerId++, at: now + delay, every: repeat ? delay : 0, fn: fn };
    timers.push(t);
    return t.id;
  }

  function runTimers() {
    var due = timers.filter(function (t) { return t.at <= now; });
    for (var i = 0; i < due.length; i++) {
      var t = due[i];
      var index = timers.indexOf(t);
      if (index < 0) continue;
      if (t.every > 0) t.at += t.every;
      else timers.splice(index, 1);
      t.fn();
    }
  }

  // mulberry32, seeded by the host so a session can be replayed and checked
  function rand() {
    seed = (seed + 0x6D2B79F5) | 0;
    var t = Math.imul(seed ^ (seed >>> 15), 1 | seed);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  }

  function deepFreeze(o) {
    Object.getOwnPropertyNames(o).forEach(function (k) {
      var d = Object.getOwnPropertyDescriptor(o, k);
      if (d.value && typeof d.value === 'object') deepFreeze(d.value);
    });
    return Object.freeze(o);
  }

  var sdk = {
    pet: makeObject(0, 0, 0),
    get time() { return now; },
    spawn: function (key, pos) {
      var id = nextId++;
      var x = num(pos && pos.x), y = num(pos && pos.y);
      emit({ op: 'spawn', id: id, key: String(key), x: x, y: y });
      return makeObject(id, x, y);
    },
    random: function (min, max) {
      var r = rand();
      return arguments.length >= 2 ? min + r * (max - min) : r;
    },
    timer: {
      after: function (sec, fn) { return addTimer(num(sec), fn, false); },
      every: function (sec, fn) { return addTimer(Math.max(num(sec), 0.016), fn, true); },
      cancel: function (id) { timers = timers.filter(function (t) { return t.id !== id; }); }
    },
    input: {
      onTap: function (fn) { tapHandlers.push(fn); },
      onDrag: function (fn) { dragHandlers.push(fn); }
    },
    ui: {
      setScore: function (n) { emit({ op: 'score', n: Math.floor(num(n)) }); },
      showTimer: function (sec) { emit({ op: 'timer', x: num(sec) }); },
      banner: function (text) { emit({ op: 'banner', s: String(text) }); }
    },
    audio: {
      play: function (key) { emit({ op: 'sound', key: String(key) }); },
      music: function (key) { emit({ op: 'music', key: String(key) }); }
    },
    camera: {
      use: function (preset) { emit({ op: 'camera', s: String(preset) }); },
      shake: function (strength) { emit({ op: 'shake', x: Math.min(Math.max(num(strength), 0), 1) }); }
    },
    log: function () {
      emit({ op: 'log', s: Array.prototype.join.call(arguments, ' ') });
    },
    endGame: function (result) {
      if (ended) return;
      emit({ op: 'end', n: Math.max(0, Math.floor(num(result && result.score))) });
      ended = true;
    }
  };

  function frame(json) {
    var input = JSON.parse(json);
    if (input.seed) seed = input.seed;
    var events = input.events || [];
    for (var i = 0; i < events.length; i++) {
      var e = events[i];
      switch (e.type) {
        case 'start':
          if (typeof globalThis.onStart === 'function') globalThis.onStart();
          break;
        case 'tap':
          tapHandlers.forEach(function (h) { h({ worldX: e.x, worldY: e.y }); });
          break;
        case 'drag':
          dragHandlers.forEach(function (h) { h({ worldX: e.x, worldY: e.y }); });
          break;
        case 'touch':
          (touchHandlers[e.id] || []).slice().forEach(function (h) { if (h.other === e.other) h.fn(); });
          break;
        case 'offscreen':
          if (offscreenHandlers[e.id]) offscreenHandlers[e.id]();
          break;
      }
    }
    now += input.dt || 0;
    runTimers();
    var out = JSON.stringify({ items: commands });
    commands = [];
    return out;
  }

  // Remove every non-standard global (Puerts' CS, puer, require, console...). Fails closed.
  var allowed = [
    'Object', 'Function', 'Array', 'Number', 'Boolean', 'String', 'Symbol', 'BigInt', 'Date', 'Promise',
    'RegExp', 'Error', 'AggregateError', 'EvalError', 'RangeError', 'ReferenceError', 'SyntaxError',
    'TypeError', 'URIError', 'JSON', 'Math', 'Reflect', 'Proxy', 'Map', 'Set', 'WeakMap', 'WeakSet',
    'WeakRef', 'FinalizationRegistry', 'ArrayBuffer', 'SharedArrayBuffer', 'DataView', 'Atomics',
    'Int8Array', 'Uint8Array', 'Uint8ClampedArray', 'Int16Array', 'Uint16Array', 'Int32Array',
    'Uint32Array', 'Float32Array', 'Float64Array', 'BigInt64Array', 'BigUint64Array', 'Iterator',
    'globalThis', 'undefined', 'NaN', 'Infinity', 'eval', 'isFinite', 'isNaN', 'parseFloat', 'parseInt',
    'decodeURI', 'decodeURIComponent', 'encodeURI', 'encodeURIComponent', 'escape', 'unescape'
  ];
  Object.getOwnPropertyNames(globalThis).forEach(function (name) {
    if (allowed.indexOf(name) >= 0) return;
    try { delete globalThis[name]; } catch (e) { }
    if (Object.prototype.hasOwnProperty.call(globalThis, name)) {
      throw new Error('sandbox: could not remove global ' + name);
    }
  });
  Object.defineProperty(globalThis, 'sdk', { value: deepFreeze(sdk), enumerable: true });

  return frame;
})()
